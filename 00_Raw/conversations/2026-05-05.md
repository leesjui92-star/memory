#### 2. `services/youtube_api.py` (YouTube API 통합 모듈)

이 모듈은 특정 검색어에 대해 구조화된 비디오 메타데이터를 가져옵니다.

```python
import os
from googleapiclient.discovery import build

class YouTubeAPIClient:
    """
    YouTube Data API v3 클라이언트를 초기화하고 핵심 기능을 제공합니다.
    """
    def __init__(self):
        api_key = os.environ.get("YOUTUBE_API_KEY")
        if not api_key:
            raise ValueError("FATAL: YOUTUBE_API_KEY가 .env 파일에 설정되지 않았습니다.")
        
        # YouTube API 서비스 객체 생성
        self.youtube = build('youtube', 'v3', developerKey=api_key)

    def search_videos(self, query: str, max_results: int = 5) -> list[dict]:
        """
        주어진 검색어에 대한 비디오 메타데이터 리스트를 가져옵니다.
        :param query: 검색할 키워드 (예: 'AI 자동화')
        :param max_results: 반환할 최대 결과 수
        :return: 구조화된 딕셔너리 목록
        """
        print(f"[API Call] Searching YouTube for '{query}'...")
        try:
            search_response = self.youtube.search().list(
                q=query,
                part='snippet',
                type='video', # 비디오만 필터링
                maxResults=max_results
            ).execute()

            videos_data = []
            for item in search_response.get('items', []):
                snippet = item['snippet']
                videos_data.append({
                    "source": "YouTube",
                    "title": snippet['title'],
                    "description_preview": snippet['description'][:100] + "...", # 설명 미리보기 (길이 제한)
                    "video_id": item['id']['videoId'],
                    "published_at": snippet['publishedAt']
                })
            return videos_data

        except Exception as e:
            print(f"!!! YouTube API 호출 중 오류 발생: {e}")
            return []
```

#### 3. `services/web_scraper.py` (웹 크롤링 모듈)

BeautifulSoup를 사용하여 주어진 URL의 핵심 텍스트 콘텐츠를 추출합니다.

```python
import requests
from bs4 import BeautifulSoup
import os

class WebScraper:
    """
    주어진 URL에서 구조화된 웹 콘텐츠(본문 텍스트)를 크롤링하여 반환합니다.
    """
    def __init__(self):
        # User-Agent 설정은 기본적인 방어 장치입니다.
        self.headers = {
            "User-Agent": os.environ.get("DEFAULT_USER_AGENT", "Custom/KnowledgeGatewayBot")
        }

    def scrape_article(self, url: str) -> dict:
        """
        URL에 접속하여 주요 기사 본문 텍스트를 추출합니다.
        :param url: 크롤링할 웹페이지 URL
        :return: 구조화된 데이터 (성공/실패 여부와 함께)
        """
        print(f"[Scrape Call] Scraping content from {url}...")
        try:
            response = requests.get(url, headers=self.headers, timeout=10)
            response.raise_for_status() # HTTP 에러 발생 시 예외 처리

            soup = BeautifulSoup(response.content, 'html.parser')
            
            # 핵심 콘텐츠 영역을 추정하여 텍스트를 추출하는 로직 (사이트마다 수정 필요)
            article_body = soup.find('article') or soup.find('main') # 가장 일반적인 기사 컨테이너 선택자 사용 시도
            
            if article_body:
                text = article_body.get_text(separator='\n', strip=True)
                return {
                    "source": "WebScrape",
                    "url": url,
                    "extracted_content": text[:2000] + "...", # 길이 제한 및 미리보기
                    "status": "SUCCESS"
                }
            else:
                # 특정 구조를 찾지 못했을 경우 전체 본문 텍스트를 시도 (차선책)
                text = soup.get_text(separator='\n', strip=True)[:2000] + "..."
                return {
                    "source": "WebScrape",
                    "url": url,
                    "extracted_content": text,
                    "status": "PARTIAL_SUCCESS_FALLBACK"
                }

        except requests.exceptions.RequestException as e:
            return {"source": "WebScrape", "url": url, "status": f"ERROR: {e}", "extracted_content": ""}
```

#### 4. `main.py` (Knowledge Ingestion Gateway 통합 로직)

이 파일은 두 서비스를 호출하고 결과를 표준 JSON 형식으로 반환하는 게이트웨이 역할을 합니다.

```python
from services.youtube_api import YouTubeAPIClient
from services.web_scraper import WebScraper
import json

class KnowledgeIngestionGateway:
    """
    YouTube API와 웹 스크래핑 기능을 통합하여 구조화된 지식 데이터 파이프라인을 구축합니다.
    """
    def __init__(self):
        print("--- 💡 KNOWLEDGE INGESTION GATEWAY 초기화 완료 ---")
        try:
            self.youtube_client = YouTubeAPIClient()
            self.scraper = WebScraper()
        except ValueError as e:
            print(f"!!! 게이트웨이 초기화 실패: {e}")
            exit(1)

    def ingest_knowledge(self, query: str, youtube_url: str = None, web_url: str = None) -> list[dict]:
        """
        단일 호출로 여러 소스의 지식을 수집하고 통합합니다.
        :param query: (필수) YouTube 검색어 또는 주제 키워드
        :param youtube_url: (선택) 특정 웹페이지 URL을 크롤링할 때 사용
        :param web_url: (선택) 실제 스크랩 할 URL
        :return: 통합된 지식 데이터 리스트
        """
        ingested_data = []

        # 1. YouTube Data API 호출 및 구조화
        youtube_results = self.youtube_client.search_videos(query=query, max_results=3)
        if youtube_results:
            ingested_data.append({"type": "YouTube", "data": youtube_results})

        # 2. 웹 크롤링 호출 및 구조화 (web_url이 제공된 경우만 실행)
        if web_url:
            scraped_result = self.scraper.scrape_article(web_url)
            ingested_data.append({"type": "Web", "data": scraped_result})

        return ingested_data

# --- 사용 예시 (API 테스트 시뮬레이션) ---
```

### 1. 스코프 정의 (Scope Definition: What to Collect)
수집 대상 지식은 **'개인의 통제 범위를 벗어난 거대 시스템의 오류 및 위험 신호'**에 초점을 맞춥니다. 정보 자체가 아니라, 그 정보가 야기하는 '불안감'과 '진단 필요성'을 추출합니다.

| 핵심 축 (Core Pillar) | 세부 주제 클러스터 (Semantic Cluster Mapping) | 주요 질문 (Diagnostic Hook) |
| :--- | :--- | :--- |
| **A. 재정적 불안 (Anti-Establishment Finance)** | 1. 연금/금융 시스템의 구조적 리스크 (물가상승률, 인구 감소 반영 실패 등). <br>2. 자산 가치 폭락 시나리오와 대비책 (디지털 자산 vs 실물 자산 배분 문제). <br>3. 세금 및 규제 변화에 따른 개인 부의 급격한 재편 가능성. | "당신의 은퇴 계획은 이 시스템 결함 앞에서도 안전합니까?" |
| **B. 사회/기술적 오류 (System Failure Exposure)** | 1. 대규모 데이터 유출 또는 프라이버시 침해 사례와 법적 공백 지점. <br>2. 거대 IT 플랫폼의 독과점 심화 및 '탈중앙화' 필요성에 대한 논쟁. <br>3. 기후 변화, 전염병 등 예측 불가능한 블랙 스완 이벤트가 경제에 미치는 영향. | "당신이 믿고 의존하는 이 시스템은 정말 안전합니까?" |
| **C. 개인 행동의 오류 (Personal Blind Spots)** | 1. 일반인이 쉽게 빠지는 재테크/투자 심리적 함정 (FOMO, Herd Mentality 등). <br>2. 자기 계발 콘텐츠에 대한 과도한 의존과 실제 성과 간의 괴리점 분석. <br>3. 사회적 성공 기준(남들이 하는 것)이 개인의 생존력에 미치는 역효과. | "당신은 왜 남들의 함정에 계속 빠져들게 되는가?" |

### 2. 수집 기준 (Collection Criteria: How to Score Data)
수집된 모든 정보는 다음 두 가지 핵심 지표를 통해 **'위험 가치 점수(Knowledge Risk Score)'**로 평가되어야 합니다.

#### ① 논쟁성 (Controversy Index - CI): 높을수록 좋음
*   **기준:** 'A' 주장과 'B' 주장이 명확하게 대립하며, 어느 쪽도 단정적으로 결론 내릴 수 없는 영역. (예: 정부 정책 vs 민간 전문가 의견의 상충).
*   **검증 방법:** 최소 3개 이상의 출처(매체, 학자 등)가 각기 다른 근거를 제시하는지 확인합니다.

#### ② 정보 밀도 및 진단 필요성 (Low Information Density & Diagnostic Need - LID): 낮을수록 좋음
*   **기준:** 단순한 사실 나열이나 명확한 정답이 없는 영역. 이 정보를 접했을 때 시청자가 "그래서 나는 무엇을 해야 하지?"라는 질문에 직면하게 만드는 주제.
*   **검증 방법:** 데이터 수집 후, 해당 지식으로 작성할 **가상의 '체크리스트/진단 항목 5가지'**를 만들 수 있는지 확인합니다. (➡️ 즉각적인 행동 유도 가능 여부)

### 3. 실행 계획 및 자동화 연동
Developer 에이전트가 구축한 `Knowledge Ingestion Gateway API`의 파이프라인을 활용하여 다음 워크플로우로 데이터를 처리합니다.

1.  **[Input]**: 선정된 키워드 기반으로 웹/YouTube 검색 및 크롤링 실행 (API 호출).

### 2. 핵심 산출물 추출 프로토콜 상세 (Extraction Protocols)

#### A. 트랜스크립트(Transcript) 구조화 요구사항
1.  **분할 단위:** 모든 문장은 반드시 시작 시간(`start`)과 종료 시간(`end`)을 포함해야 합니다.
2.  **필터링:** 단순한 구어체 반복이나 잡담은 제외하고, **주요 주장이 담긴 핵심 문장(Thesis Statement)**만 추출합니다.
3.  **메타 태그:** 각 스크립트 블록에 해당 내용이 다루는 주제 분류 코드 (예: `[FINANCE]`, `[SYSTEM_FAIL]`, `[BEHAVIORAL]`)를 필수적으로 붙입니다.

#### B. 최대 반응 30초 구간 추출 프로토콜 (`Reaction Hotspot Finder`)
이것은 단순한 시청률 지표가 아닌, **'집단적 의문점 폭발'**의 시간대를 포착하는 것이 목표입니다.

1.  **데이터 결합:** `Comment Harvester`에서 수집된 댓글 데이터 (댓글 내용 + 작성 시간)와 스크립트 타임스탬프를 교차 분석합니다.
2.  **가중치 산정:** 다음 요소들의 가중치를 조합하여 점수화합니다:
    *   **A. 감정 강도 (Sentiment Score):** "??", "이게 가능한가요?", "진짜 위험한데" 등 의문과 공포를 나타내는 키워드 비율. (최우선)
    *   **B. 밀도 (Density):** 짧은 시간 내에 많은 댓글이 쏟아져 나오는 구간. (집단적 불안감의 증거)
    *   **C. 핵심 키워드 일치:** 현재 스크립트가 언급하는 위험 주제와 댓글에서 공통적으로 추출되는 단어 간의 연관성.
3.  **결과 도출:** 이 세 가지 가중치를 종합하여 점수가 가장 높은 **최대 30초 구간 (Hotspot)**을 지정하고, 해당 시간대의 스크립트 문장과 그 구간에 집중된 대표 댓글(Representative Comment)들을 함께 추출합니다.

## 🎯 Part I. 시스템 작동 원리 요약 (The Core Philosophy)

새로운 지식 수집 시스템은 **'정보의 양'**을 모으는 것이 아니라, 정보가 야기하는 **'시청자의 불안감과 의문(Structural Anxiety)'**을 추출하는 것에 초점을 맞춥니다. 모든 과정은 궁극적으로 시청자에게 '진단 강박(Diagnostic Compulsion)'을 유발하고 다음 단계로의 연결 고리(Funnel)를 만드는 것을 목표합니다.

### 💡 핵심 원칙: 정보 $\rightarrow$ 위험 가치 점수 (Knowledge Risk Score, KRS)
모든 수집된 데이터는 다음과 같은 과정을 거쳐 지식으로 변환됩니다.

1.  **관찰:** 특정 주제나 콘텐츠(블로그 글/유튜브 영상)를 관찰합니다.
2.  **구조화:** 해당 정보가 다루는 핵심 주장을 문장 단위와 시간대별로 분리하고 분류 코드를 부여합니다. (예: `[FINANCE]`, `[SYSTEM_FAIL]`).
3.  **분석/점수화:** 이 정보가 시청자에게 던지는 **논쟁성(CI)**, **불안 요소**, 그리고 **개인이 통제할 수 없는 시스템의 결함 지점**을 평가하여 KRS를 산출합니다.

### Step 2: 데이터 정제 및 불안 요소 추출 (Researcher Protocol 적용)

2.  **Hotspot 매핑 (YouTube 한정):** 트랜스크립트 중 감성 변화가 급격하거나, 댓글에서 높은 분노/공포 지표가 발견된 **특정 시간대(30초 구간)**를 우선 추출합니다. 이 부분이 콘텐츠의 '클라이맥스'입니다.
3.  **위험 가치 스코어링:** 각 데이터 블록에 대해 다음 질문을 던집니다: "이 정보는 시청자가 자신의 삶이나 재정에 대한 근원적 의문을 갖게 할 수 있는가?" $\rightarrow$ **KRS를 부여합니다.**

*   **[KRS 점수]:** (높음/중간/낮음) - 구조적 위험도.
*   **[LOOP HOOK]:** 이 지식을 바탕으로 시청자에게 던질 수 있는 구체적인 질문 1~2가지.
*   **[ANTI-CLIMAX]:** 콘텐츠의 마지막에 제시할 '진단 프로토콜' 또는 해결책의 이름(미끼).
