```python
# c:\Users\leesj\OneDrive\Desktop\사용하지 않는 바탕화면\초보프로젝트\src\sync_validator.py

class SyncOrchestrator:
    def __init__(self, storyboard_data):
        self.storyboard = storyboard_data
        self.defects = []  # 결함 목록을 저장하는 리스트 추가

    def _validate_time_sequence(self, scene):
        """Scene 간 시간 순서 및 길이 유효성 검사"""
        start = scene['metadata']['start_time']
        end = scene['metadata']['end_time']
        duration = end - start

        if start < 0 or end < 0:
            self.defects.append(f"[{scene['id']}]: Time Code Error. Negative time detected.") # Defect logging
        elif duration <= 0:
            self.defects.append(f"[{scene['id']}]: Timing Gap/Zero Duration. Start ({start}s) >= End ({end}s).")

    def simulate_asset_call(self, scene):
        """
        [NEW] 실제 시스템 에셋 호출을 시뮬레이션하고 결함을 탐지하는 로직.
        이것은 단순히 데이터 검증을 넘어선 '시스템 통합 테스트'의 핵심입니다.
        """
        asset_id = scene['metadata'].get('visual_asset_id')

        if not asset_id:
            self.defects.append(f"[{scene['id']}]: Missing Asset ID. Visual rendering cannot proceed.")
            return False

        # 1. 가상의 '에셋 로딩 실패' 시뮬레이션 (Edge Case)
        if "ERROR_ASSET_" in asset_id:
             self.defects.append(f"[{scene['id']}]: CRITICAL ASSET FAILURE. ID '{asset_id}' not found on CDN/Server.")
             return False

        # 2. 시간 코드 불일치 시뮬레이션 (Edge Case)
        if scene['metadata'].get('transition_required') and not self._check_overlap(scene, "PreviousScene"):
            self.defects.append(f"[{scene['id']}]: TRANSITION FAILURE. Transition requested but temporal overlap check failed.")
            return False

        # 3. 권위적인 요소 누락 검증 (Business Logic Check)
        if scene['content']['type'] == 'CTA' and not any(keyword in scene['content']['text'].upper() for keyword in ["ACT NOW", "WARNING"]):
             self.defects.append(f"[{scene['id']}]: BUSINESS LOGIC GAP. CTA detected but lacks high-urgency language (e.g., ACT NOW, WARNING).")

        return True


    def run_validation(self):
        """전체 스토리보드에 대한 통합 검증을 실행합니다."""
        print("⚙️ Running Comprehensive Sync Validator...")
        for i, scene in enumerate(self.storyboard['scenes']):
            # 1. 기본 구조 유효성 검사
            self._validate_time_sequence(scene)

            # 2. 에셋 호출 시뮬레이션 (가장 중요!)
            self.simulate_asset_call(scene)
        
        print("\n✅ Validation Complete.")
        return self.defects

    def generate_patch_file(self, defects):
        """
        [NEW] 발견된 결함 목록을 기반으로 Git Patch 형식의 수정 지침서를 생성합니다.
        이는 개발팀이 바로 적용할 수 있는 '패치 패키지'가 됩니다.
        """
        if not defects:
            return "No patches required. System is fully synced.\n"

        patch_content = [
            "--- START LIVE DEBUGGING PATCH PACKAGE v1.0 ---",
            "Generated by SyncOrchestrator.",
            f"Timestamp: {datetime.now().strftime('%Y%m%d-%H%M')}",
            "\n========================================================\n",
            "# FIXES REQUIRED (Defects found):\n"
        ]

        for defect in defects:
            # Defect 메시지에서 어느 '파일'의 어느 '라인'을 수정해야 하는지 추정하여 지시.
            patch_content.append(f"\n### [Defect Found]: {defect}")
            if "Asset ID" in defect and "CRITICAL ASSET FAILURE" in defect:
                # 예외 케이스에 대한 패치 제안
                patch_content.append("--- File: assets/visuals/scene_[ID].json")
                patch_content.append("+++ New Content Required:")
                patch_content.append(f"- asset_id: {defect.split(':')[1].strip()}") # 실패한 ID 제거
                patch_content.append("+ asset_id: CORRECTLY_MAPPED_ASSET_{{scene['id']}}_V2") # 수정된 ID 삽입
            elif "Timing Gap/Zero Duration" in defect:
                 # 시간 조정에 대한 패치 제안
                patch_content.append("--- File: storyboard/timing_adjustments.py")
                patch_content.append("""def fix_duration(scene_id, current_time):""")
                patch_content.append(f"    return (current_time + 1.5) # 강제로 1.5초의 최소 버퍼 시간 추가")
            else:
                 # 기타 논리적 오류에 대한 일반적인 수정 지침
                 patch_content.append("-> Manual Review Required: Check surrounding code/metadata for logical consistency.")

        patch_content.append("\n--- END LIVE DEBUGGING PATCH PACKAGE ---")
        return "\n".join(patch_content)
```
